# Glossary

Quick definitions for terms you’ll see across this library.

export const sorted = [...glossary].sort((a, b) =>
  a.term.localeCompare(b.term),
);

<ul>
  {sorted.map((item) => (
    <li key={item.term}>
      <strong id={item.term.toLowerCase().replace(/[^a-z0-9]+/g, '-')}>
        {item.term}
      </strong>
      {' — '}
      {item.definition}
    </li>
  ))}
</ul>
